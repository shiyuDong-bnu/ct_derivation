The three-body operators surviving the space restriction are decomposed into two-body and one-body operators (plus scalar terms) with density-matrix coefficients, using the spin-free cumulant decomposition of \texttt{cumulant\_decomposition/spin\_free/spin\_free.tex}:
\begin{align}
\hat{E}^{pqr}_{stu} \approx &= D^{p}_{s} \hat{E}^{qr}_{tu} - \tfrac{1}{2}\, D^{p}_{t} \hat{E}^{qr}_{su} - \tfrac{1}{2}\, D^{p}_{u} \hat{E}^{qr}_{ts} \\
&\quad - \tfrac{1}{2}\, D^{q}_{s} \hat{E}^{pr}_{tu} + D^{q}_{t} \hat{E}^{pr}_{su} - \tfrac{1}{2}\, D^{q}_{u} \hat{E}^{pr}_{st} \\
&\quad - \tfrac{1}{2}\, D^{r}_{s} \hat{E}^{qp}_{tu} - \tfrac{1}{2}\, D^{r}_{t} \hat{E}^{pq}_{su} + D^{r}_{u} \hat{E}^{pq}_{st} + D^{pq}_{st} \hat{E}^{r}_{u} \\
&\quad - \tfrac{1}{2}\, D^{pq}_{su} \hat{E}^{r}_{t} - \tfrac{1}{2}\, D^{pq}_{ut} \hat{E}^{r}_{s} - \tfrac{1}{2}\, D^{pr}_{st} \hat{E}^{q}_{u} \\
&\quad + D^{pr}_{su} \hat{E}^{q}_{t} - \tfrac{1}{2}\, D^{pr}_{tu} \hat{E}^{q}_{s} - \tfrac{1}{2}\, D^{qr}_{su} \hat{E}^{p}_{t} + D^{qr}_{tu} \hat{E}^{p}_{s} \\
&\quad - \tfrac{1}{2}\, D^{rq}_{st} \hat{E}^{p}_{u} - 2\, D^{p}_{s} D^{q}_{t} \hat{E}^{r}_{u} - 2\, D^{p}_{s} D^{r}_{u} \hat{E}^{q}_{t} \\
&\quad - \tfrac{1}{2}\, D^{p}_{t} D^{q}_{u} \hat{E}^{r}_{s} - \tfrac{1}{2}\, D^{p}_{t} D^{r}_{s} \hat{E}^{q}_{u} - \tfrac{1}{2}\, D^{p}_{u} D^{q}_{s} \hat{E}^{r}_{t} \\
&\quad - \tfrac{1}{2}\, D^{p}_{u} D^{r}_{t} \hat{E}^{q}_{s} - \tfrac{1}{2}\, D^{q}_{s} D^{r}_{t} \hat{E}^{p}_{u} - 2\, D^{q}_{t} D^{r}_{u} \hat{E}^{p}_{s} \\
&\quad - \tfrac{1}{2}\, D^{q}_{u} D^{r}_{s} \hat{E}^{p}_{t} + D^{p}_{s} D^{q}_{u} \hat{E}^{r}_{t} + D^{p}_{s} D^{r}_{t} \hat{E}^{q}_{u} \\
&\quad + D^{p}_{t} D^{q}_{s} \hat{E}^{r}_{u} + D^{p}_{t} D^{r}_{u} \hat{E}^{q}_{s} + D^{p}_{u} D^{q}_{t} \hat{E}^{r}_{s} + D^{p}_{u} D^{r}_{s} \hat{E}^{q}_{t} \\
&\quad + D^{q}_{s} D^{r}_{u} \hat{E}^{p}_{t} + D^{q}_{t} D^{r}_{s} \hat{E}^{p}_{u} + D^{q}_{u} D^{r}_{t} \hat{E}^{p}_{s} - D^{p}_{s} D^{qr}_{tu}  \\
&\quad + \tfrac{1}{2}\, D^{p}_{t} D^{qr}_{su}  + \tfrac{1}{2}\, D^{p}_{u} D^{rq}_{st}  + \tfrac{1}{2}\, D^{q}_{s} D^{pr}_{tu}  - D^{q}_{t} D^{pr}_{su}  \\
&\quad + \tfrac{1}{2}\, D^{q}_{u} D^{pr}_{st}  + \tfrac{1}{2}\, D^{r}_{s} D^{pq}_{ut}  + \tfrac{1}{2}\, D^{r}_{t} D^{pq}_{su}  - D^{r}_{u} D^{pq}_{st}  \\
&\quad + 4\, D^{p}_{s} D^{q}_{t} D^{r}_{u}  - 2\, D^{p}_{s} D^{q}_{u} D^{r}_{t}  - 2\, D^{p}_{t} D^{q}_{s} D^{r}_{u}  + D^{p}_{t} D^{q}_{u} D^{r}_{s}  \\
&\quad + D^{p}_{u} D^{q}_{s} D^{r}_{t}  - 2\, D^{p}_{u} D^{q}_{t} D^{r}_{s} 
\end{align}
This is an operator identity: the labels $p,q,r,s,t,u$ are fixed on both sides, so no index may be renamed; only the index pairs of each operator can be permuted together (the diagonal pair-permutation symmetry of the spin-summed $\hat{E}$, no sign).  With the terms grouped by the operator in this sense, the decomposition reads
\begin{align}
\hat{E}^{pqr}_{stu} &\approx \hat{E}^{pq}_{st}\,\left(D^{r}_{u} \right) \\
&\hat{E}^{pq}_{su}\,\left(- \tfrac{1}{2}\, D^{r}_{t} \right) \\
&\hat{E}^{pq}_{ut}\,\left(- \tfrac{1}{2}\, D^{r}_{s} \right) \\
&\hat{E}^{pr}_{st}\,\left(- \tfrac{1}{2}\, D^{q}_{u} \right) \\
&\hat{E}^{pr}_{su}\,\left(D^{q}_{t} \right) \\
&\hat{E}^{pr}_{tu}\,\left(- \tfrac{1}{2}\, D^{q}_{s} \right) \\
&\hat{E}^{qr}_{su}\,\left(- \tfrac{1}{2}\, D^{p}_{t} \right) \\
&\hat{E}^{qr}_{ts}\,\left(- \tfrac{1}{2}\, D^{p}_{u} \right) \\
&\hat{E}^{qr}_{tu}\,\left(D^{p}_{s} \right) \\
&\hat{E}^{p}_{s}\,\left(D^{qr}_{tu}  - 2\, D^{q}_{t} D^{r}_{u}  + D^{q}_{u} D^{r}_{t} \right) \\
&\hat{E}^{p}_{t}\,\left(- \tfrac{1}{2}\, D^{qr}_{su}  + D^{q}_{s} D^{r}_{u}  - \tfrac{1}{2}\, D^{q}_{u} D^{r}_{s} \right) \\
&\hat{E}^{p}_{u}\,\left(- \tfrac{1}{2}\, D^{q}_{s} D^{r}_{t}  + D^{q}_{t} D^{r}_{s}  - \tfrac{1}{2}\, D^{rq}_{st} \right) \\
&\hat{E}^{q}_{s}\,\left(- \tfrac{1}{2}\, D^{pr}_{tu}  + D^{p}_{t} D^{r}_{u}  - \tfrac{1}{2}\, D^{p}_{u} D^{r}_{t} \right) \\
&\hat{E}^{q}_{t}\,\left(D^{pr}_{su}  - 2\, D^{p}_{s} D^{r}_{u}  + D^{p}_{u} D^{r}_{s} \right) \\
&\hat{E}^{q}_{u}\,\left(- \tfrac{1}{2}\, D^{pr}_{st}  + D^{p}_{s} D^{r}_{t}  - \tfrac{1}{2}\, D^{p}_{t} D^{r}_{s} \right) \\
&\hat{E}^{r}_{s}\,\left(- \tfrac{1}{2}\, D^{pq}_{ut}  - \tfrac{1}{2}\, D^{p}_{t} D^{q}_{u}  + D^{p}_{u} D^{q}_{t} \right) \\
&\hat{E}^{r}_{t}\,\left(- \tfrac{1}{2}\, D^{pq}_{su}  + D^{p}_{s} D^{q}_{u}  - \tfrac{1}{2}\, D^{p}_{u} D^{q}_{s} \right) \\
&\hat{E}^{r}_{u}\,\left(D^{pq}_{st}  - 2\, D^{p}_{s} D^{q}_{t}  + D^{p}_{t} D^{q}_{s} \right) \\
&- D^{p}_{s} D^{qr}_{tu}  + 4\, D^{p}_{s} D^{q}_{t} D^{r}_{u}  - 2\, D^{p}_{s} D^{q}_{u} D^{r}_{t}  + \tfrac{1}{2}\, D^{p}_{t} D^{qr}_{su}  \\
&\quad - 2\, D^{p}_{t} D^{q}_{s} D^{r}_{u}  + D^{p}_{t} D^{q}_{u} D^{r}_{s}  + D^{p}_{u} D^{q}_{s} D^{r}_{t}  \\
&\quad - 2\, D^{p}_{u} D^{q}_{t} D^{r}_{s}  + \tfrac{1}{2}\, D^{p}_{u} D^{rq}_{st}  + \tfrac{1}{2}\, D^{q}_{s} D^{pr}_{tu}  \\
&\quad - D^{q}_{t} D^{pr}_{su}  + \tfrac{1}{2}\, D^{q}_{u} D^{pr}_{st}  + \tfrac{1}{2}\, D^{r}_{s} D^{pq}_{ut}  \\
&\quad + \tfrac{1}{2}\, D^{r}_{t} D^{pq}_{su}  - D^{r}_{u} D^{pq}_{st} 
\end{align}
(19 operator groups; a group with no $\hat{E}$ shown is the scalar part.)
The $\approx$ drops the normal-ordered three-body remainder and the three-cumulant $\lambda^{pqr}_{stu}$; for a single-determinant reference $\lambda_3 = 0$ identically. Every density matrix carrying a virtual index vanishes; density matrices with occupied or whole-space indices are kept symbolic as $D^i_j$, $D^{ij}_{kl}$, \dots\ (no delta-function substitution).  The formula is verified numerically for a closed-shell single-determinant reference on Jordan--Wigner matrices (random index assignments with and without the virtual rule).
