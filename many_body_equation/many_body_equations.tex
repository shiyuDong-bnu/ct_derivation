All commutator blocks of the \texttt{commutator} stage, taken from its API (\texttt{commutator\_blocks()} of \texttt{commutator/generate\_commutators.py}) and expanded with the generalized Wick theorem (\texttt{generalized\_wick\_theorem/generalized\_wick\_commutator.py}). The truncation is by perturbation order (M\o ller--Plesset counting $\hat{h} \sim O(1)$, $\hat{V} \sim O(\lambda)$, $\hat{T} \sim O(\lambda)$; the Fock operator differs from $\hat{h}$ only at $O(\lambda)$), correct through $O(\lambda^2)$: the full first-order term $[\hat{H}_b, \hat{A}]$ with $\hat{H}_b = \hat{h} + \hat{V}$, and the Fock half of the second order, $\tfrac{1}{2}[[\hat{F}, \hat{A}], \hat{A}]$ (writing $\hat{F}$ for $\hat{h}$ there changes the result only at $O(\lambda^3)$).  The valence double commutator is $O(\lambda^3)$ and is not expanded. Repeated indices are summed over their space.  The index spaces and the delta rules are those of \texttt{readme.md}:
\begin{align}
p_1,p_2,\ldots &:\ \text{whole space}, &
i_1,i_2,\ldots &:\ \text{occupied}, &
a_1,a_2,\ldots &:\ \text{virtual}, \\
\delta(\text{whole},\text{vir}) &\to \text{virtual}, &
\delta(\text{whole},\text{occ}) &\to \text{occupied}, &
\delta(\text{occ},\text{vir}) &= 0.
\end{align}
The operators are
\begin{align}
\hat{h} &= h^q_p\, \hat{E}^p_q, &
\hat{V} &= \tfrac{1}{2} g^{pq}_{rs}\, \hat{E}^{rs}_{pq}, \\
\hat{F} &= f^q_p\, \hat{E}^p_q, &
\hat{T}_1 &= t^a_i\, \hat{E}^a_i, \\
\hat{T}_2 &= \tfrac{1}{2} t^{ab}_{ij}\, \hat{E}^{ab}_{ij},
\end{align}
with $\hat{h}$, $\hat{F}$ and $\hat{V}$ Hermitian.  The dummy indices are numbered per space ($a_1,a_2,\ldots$ virtual, $i_1,i_2,\ldots$ occupied, $p_1,p_2,\ldots$ whole space); the second excitation of the nested commutators carries the next pair ($a_3,a_4$ and $i_3,i_4$). The coefficients are real and follow the spin-integrated convention: no index antisymmetry is used anywhere, $g^{pq}_{rs}$ and $t^{ab}_{ij}$ carry only the simultaneous pair swap $g^{pq}_{rs} = g^{qp}_{sr}$, $t^{ab}_{ij} = t^{ba}_{ji}$, and $h^q_p = h^p_q$, $f^q_p = f^p_q$, $g^{pq}_{rs} = g^{rs}_{pq}$. Conjugate blocks are expanded by conjugating the first-order result term by term ($\hat{E}^{u\ldots}_{l\ldots} \to \hat{E}^{l\ldots}_{u\ldots}$, $f^q_p \to f^p_q$, $g^{pq}_{rs} \to g^{rs}_{pq}$; $t$ is real and unchanged). Blocks containing operators of rank higher than two are marked; their cumulant expansion (and the $[\ldots]_{1,2}$ truncation) is done separately.  The expanded blocks are available programmatically through \texttt{expanded\_blocks()}.

Equal terms are merged once per block, after the full expansion (\texttt{merge}), over every identity that is exact in this convention (\texttt{readme.md}, simplification steps 2--4): the diagonal pair-permutation symmetry of the spin-summed $\hat{E}$ (the two index rows may only be permuted together, with no sign, $\hat{E}^{qp\ldots}_{rs\ldots} = \hat{E}^{pq\ldots}_{sr\ldots}$; a single row alone is \emph{not} a symmetry of the spin-summed operator), the pair symmetries of $g$ and $t$ above, and the renaming of summed indices --- the free indices of the operator are the summation labels of the operator basis sum, so they can be renamed like tensor dummies.

Every block is verified numerically against the direct nested commutator on Jordan--Wigner matrices, twice: with spin-orbital operators (4 modes, 2 occupied $+$ 2 virtual) and with the genuinely spin-summed operators (4 spatial orbitals $=$ 8 spin orbitals); the second check is the one that sees the merging steps.

\section{First-order commutators}
\begin{align}
\left[\hat{h},\, \hat{T}_{1}\right] &= - h^{i_{1}}_{p_{1}} t^{a_{1}}_{i_{1}} \hat{E}^{a_{1}}_{p_{1}} + h^{a_{1}}_{p_{1}} t^{a_{1}}_{i_{1}} \hat{E}^{p_{1}}_{i_{1}}
\end{align}
\begin{align}
\left[\hat{h},\, \hat{T}_{2}\right] &= - h^{i_{2}}_{p_{1}} t^{a_{1}a_{2}}_{i_{1}i_{2}} \hat{E}^{a_{1}a_{2}}_{i_{1}p_{1}} \\
&\quad + h^{a_{2}}_{p_{1}} t^{a_{2}a_{1}}_{i_{2}i_{1}} \hat{E}^{a_{1}p_{1}}_{i_{1}i_{2}}
\end{align}
\begin{align}
\left[\hat{V},\, \hat{T}_{1}\right] &= - g^{i_{1}p_{1}}_{p_{2}p_{3}} t^{a_{1}}_{i_{1}} \hat{E}^{a_{1}p_{1}}_{p_{2}p_{3}} \\
&\quad + g^{a_{1}p_{3}}_{p_{1}p_{2}} t^{a_{1}}_{i_{1}} \hat{E}^{p_{1}p_{2}}_{i_{1}p_{3}}
\end{align}
\begin{align}
\left[\hat{V},\, \hat{T}_{2}\right] &= - \tfrac{1}{2}\, g^{i_{1}i_{2}}_{p_{1}p_{2}} t^{a_{1}a_{2}}_{i_{1}i_{2}} \hat{E}^{a_{1}a_{2}}_{p_{1}p_{2}} \\
&\quad + \tfrac{1}{2}\, g^{a_{1}a_{2}}_{p_{1}p_{2}} t^{a_{1}a_{2}}_{i_{1}i_{2}} \hat{E}^{p_{1}p_{2}}_{i_{1}i_{2}} \\
&\quad - g^{i_{2}p_{1}}_{p_{2}p_{3}} t^{a_{1}a_{2}}_{i_{1}i_{2}} \hat{E}^{a_{1}a_{2}p_{1}}_{i_{1}p_{2}p_{3}} \\
&\quad + g^{a_{2}p_{3}}_{p_{1}p_{2}} t^{a_{2}a_{1}}_{i_{2}i_{1}} \hat{E}^{a_{1}p_{1}p_{2}}_{i_{1}i_{2}p_{3}}\quad (\text{three-body terms})
\end{align}

\section{Second order: Fock blocks}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{1}\right],\, \hat{T}_{1}\right] &= - 2\, f^{a_{2}}_{i_{2}} t^{a_{1}}_{i_{2}} t^{a_{2}}_{i_{1}} \hat{E}^{a_{1}}_{i_{1}}
\end{align}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{1}\right],\, \hat{T}_{2}\right] &= - f^{a_{3}}_{i_{3}} t^{a_{1}a_{2}}_{i_{1}i_{3}} t^{a_{3}}_{i_{2}} \hat{E}^{a_{1}a_{2}}_{i_{1}i_{2}} \\
&\quad - f^{a_{3}}_{i_{3}} t^{a_{1}}_{i_{3}} t^{a_{3}a_{2}}_{i_{1}i_{2}} \hat{E}^{a_{1}a_{2}}_{i_{1}i_{2}}
\end{align}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{2}\right],\, \hat{T}_{1}\right] &= - f^{a_{3}}_{i_{3}} t^{a_{1}a_{2}}_{i_{1}i_{3}} t^{a_{3}}_{i_{2}} \hat{E}^{a_{1}a_{2}}_{i_{1}i_{2}} \\
&\quad - f^{a_{3}}_{i_{3}} t^{a_{1}}_{i_{3}} t^{a_{3}a_{2}}_{i_{1}i_{2}} \hat{E}^{a_{1}a_{2}}_{i_{1}i_{2}}
\end{align}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{2}\right],\, \hat{T}_{2}\right] &= - 2\, f^{a_{4}}_{i_{4}} t^{a_{1}a_{2}}_{i_{1}i_{4}} t^{a_{4}a_{3}}_{i_{2}i_{3}} \hat{E}^{a_{1}a_{2}a_{3}}_{i_{1}i_{2}i_{3}}\quad (\text{three-body terms})
\end{align}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{1}\right]^{\dagger},\, \hat{T}_{1}\right] &= f^{i_{1}}_{i_{2}} t^{a_{1}}_{i_{1}} t^{a_{2}}_{i_{2}} \hat{E}^{a_{1}}_{a_{2}} \\
&\quad - f^{a_{2}}_{p_{1}} t^{a_{1}}_{i_{1}} t^{a_{2}}_{i_{1}} \hat{E}^{a_{1}}_{p_{1}} + f^{a_{1}}_{a_{2}} t^{a_{1}}_{i_{1}} t^{a_{2}}_{i_{2}} \hat{E}^{i_{1}}_{i_{2}} \\
&\quad - f^{i_{2}}_{p_{1}} t^{a_{1}}_{i_{1}} t^{a_{1}}_{i_{2}} \hat{E}^{p_{1}}_{i_{1}}
\end{align}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{1}\right]^{\dagger},\, \hat{T}_{2}\right] &= f^{i_{2}}_{i_{3}} t^{a_{1}a_{2}}_{i_{2}i_{1}} t^{a_{3}}_{i_{3}} \hat{E}^{a_{1}a_{2}}_{a_{3}i_{1}} \\
&\quad - f^{a_{3}}_{p_{1}} t^{a_{1}a_{2}}_{i_{1}i_{2}} t^{a_{3}}_{i_{2}} \hat{E}^{a_{1}a_{2}}_{i_{1}p_{1}} \\
&\quad + f^{a_{2}}_{a_{3}} t^{a_{2}a_{1}}_{i_{3}i_{2}} t^{a_{3}}_{i_{1}} \hat{E}^{a_{1}i_{1}}_{i_{2}i_{3}} \\
&\quad - f^{i_{3}}_{p_{1}} t^{a_{2}a_{1}}_{i_{2}i_{1}} t^{a_{2}}_{i_{3}} \hat{E}^{a_{1}p_{1}}_{i_{1}i_{2}}
\end{align}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{2}\right]^{\dagger},\, \hat{T}_{1}\right] &= f^{i_{2}}_{i_{3}} t^{a_{1}}_{i_{2}} t^{a_{2}a_{3}}_{i_{3}i_{1}} \hat{E}^{a_{1}i_{1}}_{a_{2}a_{3}} \\
&\quad - f^{a_{3}}_{p_{1}} t^{a_{1}}_{i_{2}} t^{a_{3}a_{2}}_{i_{1}i_{2}} \hat{E}^{a_{1}i_{1}}_{a_{2}p_{1}} \\
&\quad - f^{a_{3}}_{p_{1}} t^{a_{1}}_{i_{2}} t^{a_{3}a_{2}}_{i_{2}i_{1}} \hat{E}^{a_{1}i_{1}}_{p_{1}a_{2}} \\
&\quad + f^{i_{1}}_{p_{1}} t^{a_{1}}_{i_{2}} t^{a_{2}a_{3}}_{i_{2}i_{1}} \hat{E}^{a_{1}p_{1}}_{a_{2}a_{3}} \\
&\quad + f^{a_{2}}_{a_{3}} t^{a_{2}a_{1}}_{i_{2}i_{1}} t^{a_{3}}_{i_{3}} \hat{E}^{i_{1}i_{2}}_{a_{1}i_{3}} \\
&\quad + f^{a_{1}}_{p_{1}} t^{a_{1}a_{2}}_{i_{2}i_{1}} t^{a_{2}}_{i_{3}} \hat{E}^{i_{1}i_{2}}_{i_{3}p_{1}} \\
&\quad - f^{i_{3}}_{p_{1}} t^{a_{2}a_{1}}_{i_{3}i_{1}} t^{a_{2}}_{i_{2}} \hat{E}^{i_{1}p_{1}}_{a_{1}i_{2}} \\
&\quad - f^{i_{3}}_{p_{1}} t^{a_{2}a_{1}}_{i_{1}i_{3}} t^{a_{2}}_{i_{2}} \hat{E}^{i_{1}p_{1}}_{i_{2}a_{1}}
\end{align}

\left[\left[\hat{F},\, \hat{T}_{2}\right]^{\dagger},\, \hat{T}_{2}\right] &= f^{i_{1}}_{i_{2}} t^{a_{1}a_{2}}_{i_{1}i_{3}} t^{a_{3}a_{4}}_{i_{2}i_{3}} \hat{E}^{a_{1}a_{2}}_{a_{3}a_{4}} \\

&\quad + f^{a_{1}}_{a_{2}} t^{a_{1}a_{3}}_{i_{1}i_{2}} t^{a_{2}a_{3}}_{i_{3}i_{4}} \hat{E}^{i_{1}i_{2}}_{i_{3}i_{4}} \\
&\quad - f^{i_{4}}_{p_{1}} t^{a_{1}a_{2}}_{i_{1}i_{4}} t^{a_{1}a_{2}}_{i_{2}i_{3}} \hat{E}^{i_{1}p_{1}}_{i_{2}i_{3}} \\
&\quad + f^{i_{3}}_{i_{4}} t^{a_{1}a_{2}}_{i_{3}i_{2}} t^{a_{3}a_{4}}_{i_{4}i_{1}} \hat{E}^{a_{1}a_{2}i_{1}}_{a_{3}i_{2}a_{4}} \\

&\quad - f^{a_{4}}_{p_{1}} t^{a_{1}a_{2}}_{i_{2}i_{3}} t^{a_{4}a_{3}}_{i_{3}i_{1}} \hat{E}^{a_{1}a_{2}i_{1}}_{i_{2}p_{1}a_{3}} \\
&\quad + f^{i_{2}}_{p_{1}} t^{a_{1}a_{2}}_{i_{3}i_{1}} t^{a_{3}a_{4}}_{i_{3}i_{2}} \hat{E}^{a_{1}a_{2}p_{1}}_{a_{3}i_{1}a_{4}} \\
&\quad + f^{a_{3}}_{a_{4}} t^{a_{3}a_{1}}_{i_{4}i_{3}} t^{a_{4}a_{2}}_{i_{2}i_{1}} \hat{E}^{a_{1}i_{1}i_{2}}_{i_{3}a_{2}i_{4}} \\

\end{align}

\section{Assembly}
With the blocks above, the transcorrelated Hamiltonian through $O(\lambda^2)$ is
\begin{align}
\left[\hat{H}_b,\, \hat{A}\right] &= \left[\hat{h},\, \hat{A}\right] + \left[\hat{V},\, \hat{A}\right], \\
\left[\hat{h},\, \hat{A}\right] &= \left[\hat{h},\, \hat{T}_1\right] + \left[\hat{h},\, \hat{T}_2\right] + \mathrm{h.c.}, \\
\left[\hat{V},\, \hat{A}\right] &= \left[\hat{V},\, \hat{T}_1\right] + \left[\hat{V},\, \hat{T}_2\right] + \mathrm{h.c.}, \\
\tfrac{1}{2}\left[\left[\hat{F},\, \hat{A}\right],\, \hat{A}\right] &= \tfrac{1}{2}\sum_{r,s=1}^{2}\left( \left[\left[\hat{F},\, \hat{T}_r\right],\, \hat{T}_s\right] + \left[\left[\hat{F},\, \hat{T}_r\right]^{\dagger},\, \hat{T}_s\right]\right) + \mathrm{h.c.},
\end{align}
the valence double commutator $\tfrac{1}{2}[[\hat{V}, \hat{A}], \hat{A}] \sim O(\lambda^3)$ being dropped by the perturbation-order truncation.
