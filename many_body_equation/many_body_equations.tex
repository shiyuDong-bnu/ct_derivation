\maketitle

All commutator blocks of the \texttt{commutator} stage, taken from its API (\texttt{commutator\_blocks()} of \texttt{commutator/generate\_commutators.py}) and expanded with the generalized Wick theorem (\texttt{generalized\_wick\_theorem/generalized\_wick\_commutator.py}). The truncation is by perturbation order (M\o ller--Plesset counting $\hat{h} \sim O(1)$, $\hat{V} \sim O(\lambda)$, $\hat{T} \sim O(\lambda)$; the Fock operator differs from $\hat{h}$ only at $O(\lambda)$), correct through $O(\lambda^2)$: the full first-order term $[\hat{H}_b, \hat{A}]$ with $\hat{H}_b = \hat{h} + \hat{V}$, and the Fock half of the second order, $\tfrac{1}{2}[[\hat{F}, \hat{A}], \hat{A}]$ (writing $\hat{F}$ for $\hat{h}$ there changes the result only at $O(\lambda^3)$).  The valence double commutator is $O(\lambda^3)$ and is not expanded. Repeated indices are summed over their space.  The index spaces and the delta rules are those of \texttt{readme.md}:
\begin{align}
p,q,r,s &:\ \text{whole space}, &
i,j,k,l &:\ \text{occupied}, &
a,b,c,d &:\ \text{virtual}, \\
\delta(\text{whole},\text{vir}) &\to \text{virtual}, &
\delta(\text{whole},\text{occ}) &\to \text{occupied}, &
\delta(\text{occ},\text{vir}) &= 0.
\end{align}
The operators are
\begin{align}
\hat{h} &= h^q_p\, \hat{E}^p_q, &
\hat{V} &= \tfrac{1}{2} g^{pq}_{rs}\, \hat{E}^{rs}_{pq}, \\
\hat{F} &= f^q_p\, \hat{E}^p_q, &
\hat{T}_1 &= t^a_i\, \hat{E}^a_i, \\
\hat{T}_2 &= \tfrac{1}{2} t^{ab}_{ij}\, \hat{E}^{ab}_{ij},
\end{align}
with $\hat{h}$, $\hat{F}$ and $\hat{V}$ Hermitian; the second excitation of the nested commutators carries the dummy indices $c,d$ and $k,l$. The coefficients are real and follow the spin-integrated convention: no index antisymmetry is used anywhere, $g^{pq}_{rs}$ and $t^{ab}_{ij}$ carry only the simultaneous pair swap $g^{pq}_{rs} = g^{qp}_{sr}$, $t^{ab}_{ij} = t^{ba}_{ji}$, and $h^q_p = h^p_q$, $f^q_p = f^p_q$, $g^{pq}_{rs} = g^{rs}_{pq}$. Conjugate blocks are expanded by conjugating the first-order result term by term ($\hat{E}^{u\ldots}_{l\ldots} \to \hat{E}^{l\ldots}_{u\ldots}$, $f^q_p \to f^p_q$, $g^{pq}_{rs} \to g^{rs}_{pq}$; $t$ is real and unchanged). Blocks containing operators of rank higher than two are marked; their cumulant expansion (and the $[\ldots]_{1,2}$ truncation) is done separately.  The expanded blocks are available programmatically through \texttt{expanded\_blocks()}.

Equal terms are merged (\texttt{readme.md}, rule 4) over every identity that is exact in this convention: renaming of any summed index within its space --- the free indices of the operator are the summation labels of the operator basis sum, so they can be renamed like tensor dummies --- the pair symmetries of $g$ and $t$ above, and the diagonal pair-permutation symmetry of the spin-summed $\hat{E}$: the two index rows may only be permuted together, with no sign ($\hat{E}^{qp\ldots}_{rs\ldots} = \hat{E}^{pq\ldots}_{sr\ldots}$); a single row alone is \emph{not} a symmetry of the spin-summed operator.

Every block is verified numerically against the direct nested commutator on Jordan--Wigner matrices, twice: with spin-orbital operators (4 modes, 2 occupied $+$ 2 virtual) and with the genuinely spin-summed operators (4 spatial orbitals $=$ 8 spin orbitals); the second check is the one that sees the merging rules.

\section{First-order commutators}
\begin{align}
\left[\hat{h},\, \hat{T}_{1}\right] &= - h^{i}_{p} t^{a}_{i} \hat{E}^{a}_{p} + h^{a}_{p} t^{a}_{i} \hat{E}^{p}_{i}
\end{align}
\begin{align}
\left[\hat{h},\, \hat{T}_{2}\right] &= - h^{j}_{p} t^{ab}_{ij} \hat{E}^{ab}_{ip} + h^{b}_{p} t^{ab}_{ij} \hat{E}^{ap}_{ij}
\end{align}
\begin{align}
\left[\hat{V},\, \hat{T}_{1}\right] &= - g^{ip}_{qr} t^{a}_{i} \hat{E}^{ap}_{qr} + g^{ar}_{pq} t^{a}_{i} \hat{E}^{pq}_{ir}
\end{align}
\begin{align}
\left[\hat{V},\, \hat{T}_{2}\right] &= - \tfrac{1}{2}\, g^{ij}_{pq} t^{ab}_{ij} \hat{E}^{ab}_{pq} + \tfrac{1}{2}\, g^{ab}_{pq} t^{ab}_{ij} \hat{E}^{pq}_{ij} \\
&\quad - g^{jp}_{qr} t^{ab}_{ij} \hat{E}^{abp}_{iqr} + g^{br}_{pq} t^{ab}_{ij} \hat{E}^{apq}_{ijr}\quad (\text{three-body terms})
\end{align}

\section{Second order: Fock blocks}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{1}\right],\, \hat{T}_{1}\right] &= - 2\, f^{b}_{j} t^{b}_{i} t^{a}_{j} \hat{E}^{a}_{i}
\end{align}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{1}\right],\, \hat{T}_{2}\right] &= - f^{c}_{k} t^{a}_{k} t^{bc}_{ji} \hat{E}^{ab}_{ij} - f^{c}_{k} t^{c}_{j} t^{ab}_{ik} \hat{E}^{ab}_{ij}
\end{align}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{2}\right],\, \hat{T}_{1}\right] &= - f^{c}_{k} t^{ab}_{ik} t^{c}_{j} \hat{E}^{ab}_{ij} - f^{c}_{k} t^{bc}_{ji} t^{a}_{k} \hat{E}^{ab}_{ij}
\end{align}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{2}\right],\, \hat{T}_{2}\right] &= - 2\, f^{d}_{l} t^{cd}_{kj} t^{ab}_{il} \hat{E}^{abc}_{ijk}\quad (\text{three-body terms})
\end{align}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{1}\right]^{\dagger},\, \hat{T}_{1}\right] &= f^{i}_{j} t^{b}_{j} t^{a}_{i} \hat{E}^{a}_{b} - f^{b}_{p} t^{b}_{i} t^{a}_{i} \hat{E}^{a}_{p} \\
&\quad + f^{a}_{b} t^{a}_{i} t^{b}_{j} \hat{E}^{i}_{j} - f^{j}_{p} t^{a}_{j} t^{a}_{i} \hat{E}^{p}_{i}
\end{align}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{1}\right]^{\dagger},\, \hat{T}_{2}\right] &= f^{j}_{k} t^{c}_{k} t^{ab}_{ji} \hat{E}^{ab}_{ci} \\
&\quad - f^{c}_{p} t^{c}_{j} t^{ab}_{ij} \hat{E}^{ab}_{ip} + f^{b}_{c} t^{c}_{i} t^{ab}_{jk} \hat{E}^{ai}_{jk} - f^{k}_{p} t^{b}_{k} t^{ab}_{ij} \hat{E}^{ap}_{ij}
\end{align}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{2}\right]^{\dagger},\, \hat{T}_{1}\right] &= f^{j}_{k} t^{bc}_{ki} t^{a}_{j} \hat{E}^{ai}_{bc} \\
&\quad - f^{c}_{p} t^{bc}_{ji} t^{a}_{j} \hat{E}^{ai}_{bp} - f^{c}_{p} t^{bc}_{ij} t^{a}_{j} \hat{E}^{ai}_{pb} + f^{i}_{p} t^{bc}_{ji} t^{a}_{j} \hat{E}^{ap}_{bc} \\
&\quad + f^{b}_{c} t^{ab}_{ij} t^{c}_{k} \hat{E}^{ij}_{ak} + f^{a}_{p} t^{ab}_{ji} t^{b}_{k} \hat{E}^{ij}_{kp} - f^{k}_{p} t^{ab}_{ik} t^{b}_{j} \hat{E}^{ip}_{aj} \\
&\quad - f^{k}_{p} t^{ab}_{ki} t^{b}_{j} \hat{E}^{ip}_{ja}
\end{align}
\begin{align}
\left[\left[\hat{F},\, \hat{T}_{2}\right]^{\dagger},\, \hat{T}_{2}\right] &= f^{i}_{j} t^{cd}_{jk} t^{ab}_{ik} \hat{E}^{ab}_{cd} \\
&\quad - f^{d}_{p} t^{cd}_{ij} t^{ab}_{ij} \hat{E}^{ab}_{cp} + f^{a}_{b} t^{ac}_{ij} t^{bc}_{kl} \hat{E}^{ij}_{kl} - f^{l}_{p} t^{ab}_{il} t^{ab}_{jk} \hat{E}^{ip}_{jk} \\
&\quad + f^{k}_{l} t^{cd}_{li} t^{ab}_{kj} \hat{E}^{abi}_{cjd} - f^{d}_{p} t^{cd}_{ki} t^{ab}_{kj} \hat{E}^{abi}_{cjp} \\
&\quad - f^{d}_{p} t^{cd}_{ik} t^{ab}_{jk} \hat{E}^{abi}_{jpc} + f^{j}_{p} t^{cd}_{kj} t^{ab}_{ki} \hat{E}^{abp}_{cid} \\
&\quad + f^{c}_{d} t^{bd}_{ij} t^{ac}_{kl} \hat{E}^{aij}_{kbl} + f^{b}_{p} t^{bc}_{ji} t^{ac}_{kl} \hat{E}^{aij}_{klp} \\
&\quad - f^{l}_{p} t^{bc}_{il} t^{ac}_{jk} \hat{E}^{aip}_{jbk} - f^{l}_{p} t^{bc}_{li} t^{ac}_{jk} \hat{E}^{aip}_{jkb}\quad (\text{three-body terms})
\end{align}

\section{Assembly}
With the blocks above, the transcorrelated Hamiltonian through $O(\lambda^2)$ is
\begin{align}
\left[\hat{H}_b,\, \hat{A}\right] &= \left[\hat{h},\, \hat{A}\right] + \left[\hat{V},\, \hat{A}\right], \\
\left[\hat{h},\, \hat{A}\right] &= \left[\hat{h},\, \hat{T}_1\right] + \left[\hat{h},\, \hat{T}_2\right] + \mathrm{h.c.}, \\
\left[\hat{V},\, \hat{A}\right] &= \left[\hat{V},\, \hat{T}_1\right] + \left[\hat{V},\, \hat{T}_2\right] + \mathrm{h.c.}, \\
\tfrac{1}{2}\left[\left[\hat{F},\, \hat{A}\right],\, \hat{A}\right] &= \tfrac{1}{2}\sum_{r,s=1}^{2}\left( \left[\left[\hat{F},\, \hat{T}_r\right],\, \hat{T}_s\right] + \left[\left[\hat{F},\, \hat{T}_r\right]^{\dagger},\, \hat{T}_s\right]\right) + \mathrm{h.c.},
\end{align}
the valence double commutator $\tfrac{1}{2}[[\hat{V}, \hat{A}], \hat{A}] \sim O(\lambda^3)$ being dropped by the perturbation-order truncation.
